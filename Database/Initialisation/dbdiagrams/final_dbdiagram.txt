// Final database vision (database Holdings_v2, schema graph_db). Paste into https://dbdiagram.io
//
//   1_holdings_pipeline.ipynb        -> holdings_clean
//   2_f13_securities_pipeline.ipynb  -> asset
//   3_openfigi_pipeline.ipynb        -> FIGI columns of asset
//   4_owner_pipeline.ipynb           -> edgar_company (edgar_company_name stays in clean_up_1: issuer matching only)
//   5_industry_pipeline.ipynb        -> issuer columns of asset, ff_industry, sic_ff
//   6_finalisation_pipeline.ipynb    -> this schema (graph_db), copied from clean_up_1 with the columns below
//
// Industry classification: asset.issuer_cik -> edgar_company.sic -> sic_ff (pick the scheme) -> ff_industry.
//   e.g. FF12:  JOIN sic_ff f ON f.sic = c.sic AND f.scheme = 12
//               JOIN ff_industry i USING (scheme, industry_number)
//
// Foreign keys: holdings_clean -> asset, holdings_clean -> edgar_company, asset.issuer_cik -> edgar_company
// (NULL = no EDGAR issuer), sic_ff -> ff_industry.
// edgar_company.sic -> sic_ff is a logical join (one sic_ff row per scheme).
//
// Materialized views (dbdiagram has no view syntax: drawn as tables, refs are logical, no foreign keys):
//   mv_owner_asset_weight  owner -> asset edges of the bipartite graph, weight = share of the owner's portfolio

Project Holdings_v2 {
  database_type: 'PostgreSQL'
  Note: '13F holdings 1999-2020: who (owner, edgar_company) holds what (asset), per report quarter (holdings_clean), classified into Fama-French industries'
}

TableGroup holdings {
  graph_db.holdings_clean
}

TableGroup securities {
  graph_db.asset
}

TableGroup companies {
  graph_db.edgar_company
}

TableGroup graph_views {
  graph_db.mv_owner_asset_weight
}

TableGroup industries {
  graph_db.ff_industry
  graph_db.sic_ff
}

Table graph_db.holdings_clean {
  holding_id            bigint             [pk, increment, note: 'synthetic key (identity), numbered in (cik, ryear, rquarter, cusip) order']
  cik                   integer            [not null, note: 'owner CIK']
  ryear                 smallint           [not null, note: 'report year (year of rdate)']
  rquarter              smallint           [not null, note: 'report quarter (quarter of rdate); CHECK 1-4']
  fdate                 date               [not null, note: 'filing date (latest of the summed sub-positions)']
  cusip                 char(9)            [not null]
  value                 numeric(20,5)      [not null, note: 'sum of the sub-positions; CHECK <> 0, finite']
  shares                numeric(21,3)      [not null, note: 'sum of the sub-positions; CHECK <> 0, finite']
  split                 boolean

  indexes {
    (cik, ryear, rquarter, cusip) [unique, name: 'uq_holdings_cik_ryear_rquarter_cusip']
    cusip               [name: 'idx_holdings_cusip']
    (ryear, rquarter)   [name: 'idx_holdings_ryear_rquarter']
  }

  Note: 'One row per owner, report quarter and security (sub-positions of a filing, e.g. per investment discretion / voting authority / other manager, summed), after CUSIP, value, amendment and ownership cleaning; only the latest report of an owner in a quarter (two reports with different rdates in one quarter: 528 owner-quarters) and only holdings whose CUSIP is in asset (~67.6M rows from 101.8M)'
}

Table graph_db.mv_owner_asset_weight {
  ryear     smallint          [not null, note: 'report year']
  rquarter  smallint          [not null, note: 'report quarter']
  owner     integer           [not null, note: 'holdings_clean.cik']
  asset     char(9)           [not null, note: 'holdings_clean.cusip']
  weight    double precision  [not null, note: 'value / sum(value) of the owner in the quarter; sums to 1 per (ryear, rquarter, owner)']

  indexes {
    (ryear, rquarter, owner, asset) [unique, name: 'uq_mv_oaw_quarter_owner_asset']
    (asset, ryear, rquarter)        [name: 'idx_mv_oaw_asset']
  }

  Note: 'MATERIALIZED VIEW over holdings_clean (value > 0): one edge per owner, asset and report quarter of the bipartite owner-security graph, rows stored in (ryear, rquarter, cik, cusip) order (~67.6M rows). The unique index allows REFRESH MATERIALIZED VIEW CONCURRENTLY'
}

Table graph_db.asset {
  cusip              varchar(9)  [pk]
  issuer_name        text
  issuer_description text
  figi_name          text        [note: 'OpenFIGI']
  market_sector      text        [note: 'OpenFIGI']
  security_type2     text        [note: 'OpenFIGI']
  issuer_cik         integer     [note: 'industry: EDGAR CIK of the issuer (industry via edgar_company.sic)']
  industry_match     text        [note: 'name+ticker / name / name+period / name+same_industry / ambiguous / unmatched']

  indexes {
    issuer_cik                                           [name: 'asset_issuer_cik_idx']
  }

  Note: 'One row per CUSIP of the SEC Official List of Section 13(f) Securities (1996-2020) (40.6k rows; 8.9k without EDGAR issuer)'
}

Table graph_db.edgar_company {
  cik             integer    [pk]
  name            text       [note: 'current EDGAR name']
  sic             varchar(4) [note: 'current SIC; NULL for placeholder 0000']
  sic_description text
  is_owner        boolean    [not null, note: 'CIK has holdings in graph_db.holdings_clean']
  is_13f_filer    boolean    [not null, note: 'CIK filed any 13F form (13F-HR, 13F-HR/A, 13F-NT, 13F-NT/A) in EDGAR']

  indexes {
    sic  [name: 'edgar_company_sic_idx']
  }

  Note: 'One row per EDGAR company: owners (13F filers, holdings CIKs) and issuers (filers with SIC or ticker) (88k rows)'
}

Table graph_db.ff_industry {
  scheme            smallint [not null, note: 'CHECK 5 / 10 / 12 / 17 / 30 / 38 / 48 / 49']
  industry_number   smallint [not null]
  short_name        text     [not null, note: 'e.g. Banks, Chips']
  industry_name     text     [not null]
  residual_category boolean  [not null, note: 'catch-all Other of the scheme (5, 10, 12, 38 only)']

  indexes {
    (scheme, industry_number) [pk]
  }

  Note: 'Fama-French industries of every scheme (209 rows; source ff_industries.csv)'
}

Table graph_db.sic_ff {
  sic             varchar(4) [not null, note: '0000-9999, same format as edgar_company.sic']
  scheme          smallint   [not null]
  industry_number smallint   [not null]

  indexes {
    (sic, scheme)             [pk]
    (scheme, industry_number) [name: 'sic_ff_industry_idx']
  }

  Note: 'SIC code -> industry in each Fama-French scheme (~60k rows; source sic_to_ff_lookup.csv). A SIC without a row in a scheme is not classified by it (FF48: 5,054 of 10,000 SIC codes listed)'
}

Ref: graph_db.sic_ff.(scheme, industry_number) > graph_db.ff_industry.(scheme, industry_number)   // foreign key
Ref: graph_db.holdings_clean.cik > graph_db.edgar_company.cik       // foreign key, owner
Ref: graph_db.asset.issuer_cik > graph_db.edgar_company.cik         // foreign key, issuer (NULL allowed)
Ref: graph_db.edgar_company.sic <> graph_db.sic_ff.sic              // logical, many-to-many: one sic_ff row per scheme
Ref: graph_db.holdings_clean.cusip > graph_db.asset.cusip        // foreign key, security
Ref: graph_db.mv_owner_asset_weight.owner > graph_db.edgar_company.cik  // logical (view), owner
Ref: graph_db.mv_owner_asset_weight.asset > graph_db.asset.cusip        // logical (view), security
